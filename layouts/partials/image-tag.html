{{- $page := .Page -}} {{- $src := .Src -}} {{- $resolvedSrc := $src -}} {{- $imgCfg := $page.Site.Params.image -}} {{- $width := "" -}} {{- $height := "" -}} {{- /* Try to resolve as page resource */ -}} {{- $img := $page.Resources.GetMatch (strings.TrimPrefix "./" $src) -}} {{- if not $img -}} {{- $img = $page.Resources.GetMatch (printf "*%s" (path.Base $src)) -}} {{- end -}} {{- with $img -}} {{- $resolvedSrc = .RelPermalink -}} {{- end -}} {{- /* Skip SVG — dimensions may be unreliable (0 or incorrect) */ -}} {{- $isSvg := or (and $img (eq $img.MediaType.SubType "svg")) (eq (lower (path.Ext $src)) ".svg") -}} {{- if and $img (not $isSvg) -}} {{- $width = $img.Width -}} {{- $height = $img.Height -}} {{- end -}} {{- $enhanced := and $imgCfg.enabled $imgCfg.loading (not $page.IsHome) (not $isSvg) -}} {{- $border := $imgCfg.border -}} {{- if eq $page.Type "gallery" -}} {{- $border = $imgCfg.galleryBorder -}} {{- end -}} {{- $classes := slice -}} {{- if $enhanced -}}{{- $classes = $classes | append "loading-img" -}}{{- end -}} {{- if and $imgCfg.enabled $border (not $isSvg) -}}{{- $classes = $classes | append "image-bordered" -}}{{- end -}} {{- $sizeKnown := and $width $height -}} {{- if $enhanced }} {{- end }} {{- if $enhanced }} {{ partial "icon.html" (dict "name" "image") }} 点击重试 {{- end }}
