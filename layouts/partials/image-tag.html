{{- $page := .Page -}} {{- $src := .Src -}} {{- $width := "" -}} {{- $height := "" -}} {{- /* Try to resolve as page resource */ -}} {{- $img := $page.Resources.GetMatch (strings.TrimPrefix "./" $src) -}} {{- if not $img -}} {{- $img = $page.Resources.GetMatch (printf "*%s" (path.Base $src)) -}} {{- end -}} {{- /* Skip SVG — dimensions may be unreliable (0 or incorrect) */ -}} {{- $isSvg := and $img (eq $img.MediaType.SubType "svg") -}} {{- if and $img (not $isSvg) -}} {{- $width = $img.Width -}} {{- $height = $img.Height -}} {{- end -}}
