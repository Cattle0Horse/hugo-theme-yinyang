{{- $supported := slice "indigo" "violet" "teal" "amber" "rose" "graphite" -}} {{- $names := dict "indigo" "靛蓝" "violet" "紫罗兰" "teal" "青绿" "amber" "琥珀" "rose" "玫瑰" "graphite" "石墨" -}} {{- $available := slice "indigo" -}} {{- $default := "indigo" -}} {{- $allowUserChoice := false -}} {{- with .Site.Params.colorScheme -}} {{- with .available -}} {{- $configured := slice -}} {{- range . -}} {{- $name := lower (printf "%v" .) -}} {{- if and (in $supported $name) (not (in $configured $name)) -}} {{- $configured = $configured | append $name -}} {{- end -}} {{- end -}} {{- if gt (len $configured) 0 -}} {{- $available = $configured -}} {{- $default = index $available 0 -}} {{- end -}} {{- end -}} {{- with .default -}} {{- $candidate := lower (printf "%v" .) -}} {{- if in $available $candidate -}} {{- $default = $candidate -}} {{- end -}} {{- end -}} {{- $allowUserChoice = and .allowUserChoice (gt (len $available) 1) -}} {{- end -}} {{- return (dict "available" $available "names" $names "default" $default "allowUserChoice" $allowUserChoice ) -}}
